const express = require('express'); const path = require('path'); const app = express(); const PORT = 3000; app.use(express.json()); app.use(express.static(path.join(__dirname, 'public'))); // API: Receive highlighted text from client, echo back with metadata app.post('/api/speak', (req, res) => { const { text, voice, rate, pitch } = req.body; if (!text || typeof text !== 'string') { return res.status(400).json({ error: 'No text provided' }); } const sanitized = text.trim().slice(0, 5000); const wordCount = sanitized.split(/\s+/).filter(Boolean).length; const estimatedDuration = Math.ceil((wordCount / (rate || 1)) * 0.4); console.log(`[TTS] Speaking ${wordCount} words: "${sanitized.slice(0, 60)}..."`); res.json({ text: sanitized, wordCount, estimatedSeconds: estimatedDuration, voice: voice || 'default', rate: rate || 1, pitch: pitch || 1, timestamp: new Date().toISOString() }); }); // API: Get server status app.get('/api/status', (req, res) => { res.json({ status: 'ok', uptime: process.uptime() }); }); app.listen(PORT, () => { console.log(`\n🔊 Text-to-Speech Server running at http://localhost:${PORT}\n`); });
